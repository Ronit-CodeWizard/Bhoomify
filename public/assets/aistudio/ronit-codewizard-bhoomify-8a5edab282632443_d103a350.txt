import { useState, useEffect } from 'react';

export default function App() {
  const [isUnlocked, setIsUnlocked] = useState<boolean>(() => {
    const saved = sessionStorage.getItem('aura_unlocked');
    const savedTime = sessionStorage.getItem('aura_unlocked_time');
    if (saved === 'true' && savedTime) {
      const elapsed = Date.now() - parseInt(savedTime, 10);
      if (elapsed < 10 * 60 * 1000) {
        return true;
      }
    }
    sessionStorage.removeItem('aura_unlocked');
    sessionStorage.removeItem('aura_unlocked_time');
    return false;
  });

  const handleUnlock = () => {
    sessionStorage.setItem('aura_unlocked', 'true');
    sessionStorage.setItem('aura_unlocked_time', Date.now().toString());
    setIsUnlocked(true);
  };

  const player = useAudioPlayer();
  const {
    tracks,
    currentTrack,
    state,
    playError,
    togglePlay,
    playTrack,
    next,
    prev,
    seek,
    setVolume,
    toggleMute,
    toggleShuffle,
    toggleRepeat,
    toggleLike,
    reorderPlaylist,
    addToQueue,
    removeFromQueue,
  } = player;

  const { isPlaying, currentTime, volume, isMuted } = state;

  const [activeTab, setActiveTab] = useState<TabType>('now-playing');
  const [rightPanelTab, setRightPanelTab] = useState<TabType>('playlist');
  const [showShortcutOverlay, setShowShortcutOverlay] = useState(false);

  // Global Keyboard Shortcuts
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      // Ignore shortcuts if the user is typing in search input
      if (document.activeElement?.tagName === 'INPUT') return;

      switch (e.code) {
        case 'Space':
          e.preventDefault();
          togglePlay();
          break;
        case 'ArrowRight':
          e.preventDefault();
          seek(currentTime + 5);
          break;
        case 'ArrowLeft':
          e.preventDefault();
          seek(currentTime - 5);
          break;
        case 'ArrowUp':
          e.preventDefault();
          setVolume(Math.min(1, volume + 0.1));
          break;
        case 'ArrowDown':
          e.preventDefault();
          setVolume(Math.max(0, volume - 0.1));
          break;
        case 'KeyM':
          toggleMute();
          break;
        case 'KeyN':
          next();
          break;
        case 'KeyP':
          prev();
          break;
        case 'KeyL':
          toggleLike(currentTrack?.id);
          break;
        default:
          break;
      }
    };

    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [togglePlay, seek, currentTime, setVolume, volume, toggleMute, next, prev, toggleLike, currentTrack]);

  // Mood filter utility for Home screen
  const playMood = (moodType: 'chill' | 'energy' | 'focus') => {
    let targetIndex = 0;
    if (moodType === 'chill') {
      // Find Saiyaara or Ishq Mubarak
      targetIndex = tracks.findIndex((t) => t.id === 'saiyaara' || t.id === 'ishq-mubarak');
    } else if (moodType === 'energy') {
      // Find Tum Ho Toh or Nazar Na Lag Jaaye
      targetIndex = tracks.findIndex((t) => t.id === 'tum-ho-toh' || t.id === 'nazar-na-lag-jaaye');
    } else if (moodType === 'focus') {
      // Find Chahun Main Ya Naa or Jeena Jeena
      targetIndex = tracks.findIndex((t) => t.id === 'chahun-main-ya-naa' || t.id === 'jeena-jeena');
    }
    if (targetIndex !== -1) {
      playTrack(targetIndex);
    }
  };

  const renderTabContent = (tab: TabType) => {
    switch (tab) {
      case 'home':
        return (
          <div id="home-dashboard" className="flex flex-col h-full text-white px-5 pt-6 pb-40 overflow-y-auto custom-scrollbar">
            <div className="mb-6">
              <span className="text-[10px] font-bold tracking-widest text-purple-400 uppercase bg-purple-500/10 border border-purple-500/20 px-2.5 py-1 rounded-full">
                Discovery Hub
              </span>
              <h2 className="text-2xl font-bold font-display tracking-tight mt-2.5">
                Hello, Rohan
              </h2>
              <p className="text-xs text-white/40 mt-1">Ready for some modern premium sounds?</p>
            </div>

            {/* Premium Mood Selectors */}
            <h3 className="text-xs font-bold text-white/40 uppercase tracking-widest mb-3.5 flex items-center gap-1.5">
              <Compass className="w-3.5 h-3.5 text-purple-400" />
              Tune Your Mood
            </h3>
            <div className="grid grid-cols-3 gap-2.5 mb-6">
              <div
                onClick={() => playMood('chill')}
                className="bg-gradient-to-br from-violet-900/40 to-black/40 border border-violet-500/10 hover:border-violet-500/30 p-4 rounded-2xl flex flex-col items-center justify-center text-center cursor-pointer transition-all duration-300 group"
              >
                <div className="w-9 h-9 rounded-xl bg-violet-500/10 flex items-center justify-center mb-2.5 group-hover:scale-110 transition-transform">
                  <Leaf className="w-4 h-4 text-violet-400" />
                </div>
                <span className="text-xs font-bold text-white">Chill</span>
                <span className="text-[9px] text-white/40 mt-1">Ambient Beats</span>
              </div>

              <div
                onClick={() => playMood('energy')}
                className="bg-gradient-to-br from-pink-900/40 to-black/40 border border-pink-500/10 hover:border-pink-500/30 p-4 rounded-2xl flex flex-col items-center justify-center text-center cursor-pointer transition-all duration-300 group"

                  <Flame className="w-4 h-4 text-pink-400" />
                </div>

    if (!isUnlocked) return;

      const savedTime = sessionStorage.getItem('aura_unlocked_time');
      if (savedTime) {
        const elapsed = Date.now() - parseInt(savedTime, 10);
        if (elapsed >= 10 * 60 * 1000) {
          // Pause music before locking
          if (isPlaying) {
            togglePlay();
          }
          setIsUnlocked(false);
          sessionStorage.removeItem('aura_unlocked');
          sessionStorage.removeItem('aura_unlocked_time');
        }
      } else {
        setIsUnlocked(false);
      }
    }, 1000);

    return () => clearInterval(interval);
  }, [isUnlocked, isPlaying, togglePlay]);

  if (!isUnlocked) {
    return <PasswordGate onUnlock={handleUnlock} />;
  }

  return (
    <div className="min-h-screen bg-neutral-950 flex flex-col items-center justify-center p-4 relative overflow-hidden font-sans select-none antialiased">
      {/* Background radial atmosphere color blobs */}
      <div className="absolute top-[-10%] left-[-10%] w-[60%] h-[60%] rounded-full bg-purple-900/20 blur-[140px] pointer-events-none" />
      <div className="absolute bottom-[-10%] right-[-10%] w-[60%] h-[60%] rounded-full bg-pink-900/10 blur-[140px] pointer-events-none" />
      <div className="absolute top-[35%] left-[45%] w-[45%] h-[45%] rounded-full bg-violet-800/10 blur-[120px] pointer-events-none animate-pulse" />

      {/* Keyboard cheat-sheet floating button */}
      <button
        onClick={() => setShowShortcutOverlay(!showShortcutOverlay)}
        className="fixed top-4 right-4 z-50 bg-black/40 border border-white/10 hover:border-white/20 p-2.5 rounded-xl cursor-pointer hover:scale-105 active:scale-95 transition-all text-white/60 hover:text-white flex items-center gap-2 text-xs shadow-md"
        title="Show Keyboard Shortcuts"
      >
        <Keyboard className="w-4 h-4" />
        <span className="hidden sm:inline font-semibold">Shortcuts</span>
      </button>

      {/* Main layout container (dual panel on desktop, single mockup panel on mobile) */}
      <div className="w-full max-w-5xl flex items-center justify-center gap-10 z-10 h-auto">
        
        {/* DESKTOP ONLY: Left Side Now Playing Panel (Fixed device) */}
        <div className="hidden lg:block w-[390px] h-[760px] rounded-[48px] bg-black/60 border border-white/15 shadow-[0_30px_70px_rgba(0,0,0,0.8)] relative overflow-hidden">
          {/* Glass Gradient Sheet top-half background */}
          <div className="absolute top-0 left-0 right-0 h-[45%] bg-gradient-to-b from-purple-900/35 via-pink-900/10 to-transparent pointer-events-none" />

          {/* Core Player Content */}
          <VinylPlayer
            currentTrack={currentTrack}
            tracks={tracks}
            state={state}
            playError={playError}
            togglePlay={togglePlay}
            next={next}
            prev={prev}
            seek={seek}
            toggleShuffle={toggleShuffle}
            toggleRepeat={toggleRepeat}

          />

          {/* Navigation bar fixed on left panel */}

          {/* Glass Gradient Sheet top-half background */}
          <div className="absolute top-0 left-0 right-0 h-[40%] bg-gradient-to-b from-purple-950/20 via-neutral-950/10 to-transparent pointer-events-none" />

              >
                Close Panel
              </button>
            </motion.div>
          </div>
        )}
      </AnimatePresence>
    </div>
  );
}

import { useEffect, useRef, useState } from 'react';
import { motion } from 'motion/react';
import { Sparkles, Sliders, Play, Volume2 } from 'lucide-react';
import { PlayerState, Track } from '../types';

interface EqualizerScreenProps {
  state: PlayerState;
  currentTrack: Track;
  analyserNode: AnalyserNode | null;
}

type VisualizerStyle = 'bars' | 'wave' | 'radial';
type EqPreset = 'Flat' | 'Bass Boost' | 'Electronic' | 'Acoustic' | 'Vocal';

export function EqualizerScreen({ state, currentTrack, analyserNode }: EqualizerScreenProps) {
  const { isPlaying, volume } = state;
  const canvasRef = useRef<HTMLCanvasElement | null>(null);
  const [style, setStyle] = useState<VisualizerStyle>('bars');
  const [preset, setPreset] = useState<EqPreset>('Bass Boost');

  // Animation frame reference
  const animationRef = useRef<number | null>(null);

  // Equalizer presets scaling factors for visualizer
  const getPresetMultiplier = (freqIndex: number, totalBands: number) => {
    switch (preset) {
      case 'Bass Boost':
        // Boost low frequencies (left side)
        return freqIndex < totalBands * 0.25 ? 1.6 : 0.8;
      case 'Electronic':
        // Boost low and high, scoop mid
        if (freqIndex < totalBands * 0.2 || freqIndex > totalBands * 0.8) return 1.4;
        return 0.7;
      case 'Acoustic':
        // Smooth mid boost
        const midPoint = totalBands / 2;
        const distFromMid = Math.abs(freqIndex - midPoint) / midPoint;
        return 1.4 - distFromMid * 0.8;
      case 'Vocal':
        // Boost high-mids
        if (freqIndex > totalBands * 0.4 && freqIndex < totalBands * 0.7) return 1.5;
        return 0.6;
      case 'Flat':
      default:
        return 1.0;
    }
  };

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;

    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    // Handle high DPI displays
    const dpr = window.devicePixelRatio || 1;
    const rect = canvas.getBoundingClientRect();
    canvas.width = rect.width * dpr;
    canvas.height = rect.height * dpr;
    ctx.scale(dpr, dpr);

    const width = rect.width;
    const height = rect.height;

    // Simulate 40 frequency bands
    const bandCount = style === 'bars' ? 32 : 64;
    const frequencies = new Array(bandCount).fill(0);
    const targetFrequencies = new Array(bandCount).fill(0);
    
    // Wave variables
    let phase = 0;

    const render = () => {
      ctx.clearRect(0, 0, width, height);

      // Smooth interpolation values
      const smoothFactor = 0.15;

      // Update frequency bands
      let dataArray: Uint8Array | null = null;
      let bufferLength = 0;
      if (analyserNode && isPlaying) {
        bufferLength = analyserNode.frequencyBinCount;
        dataArray = new Uint8Array(bufferLength);
        analyserNode.getByteFrequencyData(dataArray);
      }

      for (let i = 0; i < bandCount; i++) {
        if (dataArray) {
          // Map to LOWER 80% of frequency spectrum where actual audible music resides (avoiding high-freq static)
          const percent = i / bandCount;
          const dataIndex = Math.floor(percent * bufferLength * 0.85);
          const rawValue = dataArray[dataIndex] || 0;

          const amplitudeMultiplier = getPresetMultiplier(i, bandCount);
          // Scale from [0, 255] to canvas height
          let targetVal = (rawValue / 255) * height * 0.85 * volume * amplitudeMultiplier;
          targetVal = Math.max(4, targetVal); // Keep small active baseline
          targetFrequencies[i] = targetVal;
        } else if (isPlaying) {
          // Generate realistic floating noise based on time, index, and volume
          const noiseFreq = 0.05 + (i * 0.02);
          const time = Date.now() * 0.005;
          const amplitudeMultiplier = getPresetMultiplier(i, bandCount);
          
          // Realistic audio wave simulation with octave noise
          const baseNoise = Math.sin(time * 0.8 + i * 0.3) * 0.3 + 0.4;
          const details = Math.sin(time * 2.3 + i * 0.9) * 0.15;
          const bassPunch = i < bandCount * 0.2 && Math.sin(time * 4) > 0.8 ? 0.3 : 0;

          let targetVal = (baseNoise + details + bassPunch) * height * 0.65 * volume * amplitudeMultiplier;
          targetVal = Math.max(4, targetVal); // Keep a small active baseline
          targetFrequencies[i] = targetVal;
        } else {
          // Flatten when paused
          targetFrequencies[i] = 4 + Math.sin(Date.now() * 0.001 + i * 0.5) * 2;
        }

        // Lerp to target
        frequencies[i] += (targetFrequencies[i] - frequencies[i]) * smoothFactor;
      }

      phase += isPlaying ? 0.08 : 0.02;

      // Render Styles
      if (style === 'bars') {
        const barWidth = width / bandCount;
        const gap = 2.5;

        for (let i = 0; i < bandCount; i++) {
          const barHeight = frequencies[i];
          const x = i * barWidth;
          const y = height - barHeight - 10;

          // Multi-stop neon gradient: Purple -> Pink -> Cyan
          const gradient = ctx.createLinearGradient(x, height, x, y);
          gradient.addColorStop(0, 'rgba(168, 85, 247, 0.2)'); // Violet/Purple
          gradient.addColorStop(0.5, 'rgba(236, 72, 153, 0.85)'); // Hot Pink
          gradient.addColorStop(1, 'rgba(0, 229, 255, 1)'); // Electric Cyan

          ctx.fillStyle = gradient;
          
          // Draw rounded bar top manually to be ultra premium
          ctx.beginPath();
          const r = Math.min(barWidth / 2 - gap / 2, 4);
          const w = barWidth - gap;
          ctx.roundRect(x + gap / 2, y, w, barHeight, [r, r, 0, 0]);
          ctx.fill();

          // Mirror top dots
          if (isPlaying && i % 2 === 0) {
            ctx.fillStyle = 'rgba(0, 229, 255, 0.6)';
            ctx.beginPath();
            ctx.arc(x + barWidth / 2, y - 6, 1.5, 0, 2 * Math.PI);
            ctx.fill();
          }
        }
      } else if (style === 'wave') {
        // High-fidelity smooth sine oscilloscopes (multiple overlapping transparent curves)
        const layers = 3;
        for (let l = 0; l < layers; l++) {
          ctx.beginPath();
          ctx.lineWidth = l === 0 ? 3.5 : 1.5;
          
          // Shift opacity and colors for each layer
          if (l === 0) ctx.strokeStyle = 'rgba(0, 229, 255, 0.95)'; // Cyan
          else if (l === 1) ctx.strokeStyle = 'rgba(236, 72, 153, 0.6)'; // Pink
          else ctx.strokeStyle = 'rgba(168, 85, 247, 0.4)'; // Purple

          const speedMultiplier = (l + 1) * 0.4;
          const amplitudeFactor = (3 - l) * 0.35;

          ctx.moveTo(0, height / 2);

          for (let x = 0; x < width; x += 5) {
            // Sample frequency visual state to deform the wave dynamically where the beat is active
            const freqIndex = Math.floor((x / width) * bandCount);
            const freqVal = frequencies[freqIndex] || 10;
            
            const sinY = Math.sin(x * 0.015 - phase * speedMultiplier) * freqVal * amplitudeFactor;
            const cosY = Math.cos(x * 0.008 + phase * speedMultiplier * 0.5) * 15;
            
            ctx.lineTo(x, height / 2 + sinY + cosY);
          }

          ctx.stroke();
        }

        // Render ambient floating nodes
        if (isPlaying) {
          ctx.fillStyle = 'rgba(236, 72, 153, 0.2)';
          for (let i = 0; i < 5; i++) {
            const nodeX = (width / 6) * (i + 1) + Math.sin(phase + i) * 15;
            const nodeY = height / 2 + Math.cos(phase * 1.2 + i) * 40;
            ctx.beginPath();
            ctx.arc(nodeX, nodeY, 4 + i, 0, 2 * Math.PI);
            ctx.fill();
          }
        }
      } else if (style === 'radial') {
        // Radial pulsating ring visualizer
        const centerX = width / 2;
        const centerY = height / 2;
        const baseRadius = 65;

        // Draw multiple rings representing frequencies
        ctx.beginPath();
        ctx.arc(centerX, centerY, baseRadius, 0, 2 * Math.PI);
        ctx.strokeStyle = 'rgba(255, 255, 255, 0.05)';
        ctx.lineWidth = 1;
        ctx.stroke();

        // Pulsating glowing core
        const pulseAmt = isPlaying ? frequencies[4] * 0.2 : 5;
        const radialGradient = ctx.createRadialGradient(centerX, centerY, 5, centerX, centerY, baseRadius + pulseAmt);
        radialGradient.addColorStop(0, 'rgba(168, 85, 247, 0.35)');
        radialGradient.addColorStop(0.5, 'rgba(236, 72, 153, 0.1)');
        radialGradient.addColorStop(1, 'rgba(0, 0, 0, 0)');
        
        ctx.fillStyle = radialGradient;
        ctx.beginPath();
        ctx.arc(centerX, centerY, baseRadius + pulseAmt, 0, 2 * Math.PI);
        ctx.fill();

        // Draw radial spikes
        const spikes = 60;
        for (let i = 0; i < spikes; i++) {
          const angle = (i / spikes) * Math.PI * 2 + phase * 0.15;
          const freqIndex = Math.floor((i / spikes) * bandCount);
          const val = frequencies[freqIndex] * 0.65;

          const startX = centerX + Math.cos(angle) * baseRadius;
          const startY = centerY + Math.sin(angle) * baseRadius;
          const endX = centerX + Math.cos(angle) * (baseRadius + val);
          const endY = centerY + Math.sin(angle) * (baseRadius + val);

          const gradient = ctx.createLinearGradient(startX, startY, endX, endY);
          gradient.addColorStop(0, 'rgba(236, 72, 153, 0.6)');
          gradient.addColorStop(1, 'rgba(0, 229, 255, 1)');

          ctx.strokeStyle = gradient;
          ctx.lineWidth = 2.5;
          ctx.beginPath();
          ctx.moveTo(startX, startY);
          ctx.lineTo(endX, endY);
          ctx.stroke();
        }
      }

      animationRef.current = requestAnimationFrame(render);
    };

    render();

    return () => {
      if (animationRef.current) {
        cancelAnimationFrame(animationRef.current);
      }
    };
  }, [style, preset, isPlaying, volume, analyserNode]);

  return (
    <div id="equalizer-screen" className="flex flex-col h-full text-white px-5 pt-6 pb-40">
      {/* Equalizer Title */}
      <div className="flex items-center justify-between mb-4">
        <div>
          <h2 className="text-xl font-bold font-display tracking-tight text-white/90">Acoustic Spectrum</h2>
          <p className="text-xs text-white/40 mt-1">Real-time spectral visualization</p>
        </div>
        <div className="flex gap-1.5 bg-white/5 p-1 rounded-xl border border-white/5">
          {(['bars', 'wave', 'radial'] as VisualizerStyle[]).map((vStyle) => (
            <button
              key={vStyle}
              onClick={() => setStyle(vStyle)}
              className={`text-xs px-2.5 py-1.5 font-bold uppercase tracking-wider rounded-lg transition-all cursor-pointer ${
                style === vStyle
                  ? 'bg-purple-500/25 text-purple-300 border border-purple-500/20'
                  : 'text-white/40 hover:text-white/70 hover:bg-white/5'
              }`}
            >
              {vStyle}
            </button>
          ))}
        </div>
      </div>

      {/* Canvas Window Container */}
      <div className="relative flex-1 bg-black/35 backdrop-blur-xl border border-white/5 rounded-3xl overflow-hidden shadow-inner flex items-center justify-center p-2 mb-4">
        {/* Track Title Watermark overlay */}
        <div className="absolute top-4 left-4 flex items-center gap-2.5 text-white/30 z-15 bg-black/20 backdrop-blur-md px-3 py-1.5 rounded-full border border-white/5">
          <Volume2 className="w-3.5 h-3.5" />
          <span className="text-[10px] font-bold tracking-widest uppercase">
            {isPlaying ? `Playing: ${currentTrack?.title}` : 'Spectral Idle'}
          </span>
        </div>

        <canvas ref={canvasRef} className="w-full h-full block" />
      </div>

      {/* Presets Grid */}
      <div className="bg-white/5 border border-white/5 rounded-2xl p-4">
        <h3 className="text-xs font-bold text-white/50 uppercase tracking-widest mb-3 flex items-center gap-1.5">
          <Sliders className="w-3.5 h-3.5 text-purple-400" />
          DSP Equalizer Presets
        </h3>

  state: PlayerState;
  playTrack: (index: number) => void;
  togglePlay: () => void;
  reorderPlaylist: (startIndex: number, endIndex: number) => void;
  toggleLike: (trackId: string) => void;
  addToQueue: (trackId: string) => void;
  removeFromQueue: (trackId: string) => void;
}

  tracks,
  currentTrack,
  state,
  playTrack,
  togglePlay,
  reorderPlaylist,
  toggleLike,
  addToQueue,
  removeFromQueue,

  const [searchQuery, setSearchQuery] = useState('');
  const [draggedIndex, setDraggedIndex] = useState<number | null>(null);
  const [activeMenuTrackId, setActiveMenuTrackId] = useState<string | null>(null);

  // Search filter
  const filteredTracks = tracks.map((track, originalIndex) => ({ track, originalIndex }))
    .filter(({ track }) => {
      const query = searchQuery.toLowerCase();
      return (
        track.title.toLowerCase().includes(query) ||
        track.artist.toLowerCase().includes(query)
      );
    });

  // Reordering handlers

    setDraggedIndex(index);
    e.dataTransfer.effectAllowed = 'move';

    e.preventDefault();
  };

  const handleDrop = (e: React.DragEvent, targetIndex: number) => {
    e.preventDefault();
    if (draggedIndex !== null && draggedIndex !== targetIndex) {
      reorderPlaylist(draggedIndex, targetIndex);
    }
    setDraggedIndex(null);
  };

  const toggleActionMenu = (trackId: string, e: React.MouseEvent) => {
    e.stopPropagation();
    setActiveMenuTrackId(activeMenuTrackId === trackId ? null : trackId);
  };

  const radius = 100;
  const strokeWidth = 3;
  const circumference = 2 * Math.PI * radius;
  const progress = duration > 0 ? currentTime / duration : 0;
  const strokeDashoffset = circumference - progress * circumference;

  const prevTrackIndex = tracks.length > 0 ? (currentTrackIndex - 1 + tracks.length) % tracks.length : 0;
  const nextTrackIndex = tracks.length > 0 ? (currentTrackIndex + 1) % tracks.length : 0;

    if (!progressRingRef.current || duration === 0) return;

    const rect = progressRingRef.current.getBoundingClientRect();
    const centerX = rect.left + rect.width / 2;
    const centerY = rect.top + rect.height / 2;

    const clientY = 'touches' in e ? e.touches[0].clientY : e.clientY;

    const dx = clientX - centerX;
    const dy = clientY - centerY;

    // Calculate angle in radians: from -PI to PI
    // We adjust it so 12 o'clock is 0 or -PI/2
    let angle = Math.atan2(dy, dx) + Math.PI / 2;
    if (angle < 0) angle += 2 * Math.PI;

  const [state, setState] = useState<PlayerState>(() => {
    const savedLiked = localStorage.getItem('aura_liked');
    const savedVolume = localStorage.getItem('aura_volume');
    const savedShuffle = localStorage.getItem('aura_shuffle');
    const savedRepeat = localStorage.getItem('aura_repeat');
    return {
      isPlaying: false,
      currentTime: 0,
      duration: 180, // Fallback duration
      volume: savedVolume ? parseFloat(savedVolume) : 0.8,
      isMuted: false,
      isShuffle: savedShuffle === 'true',
      isRepeat: (savedRepeat as 'none' | 'all' | 'one') || 'all',
      currentTrackIndex: 0,
      likedTrackIds: savedLiked ? JSON.parse(savedLiked) : [],
      queue: [],
    };
  });

  useEffect(() => {
    localStorage.setItem('aura_shuffle', state.isShuffle.toString());
  }, [state.isShuffle]);

  useEffect(() => {
    localStorage.setItem('aura_repeat', state.isRepeat);
  }, [state.isRepeat]);

  const [playError, setPlayError] = useState<string | null>(null);
  const audioRef = useRef<HTMLAudioElement | null>(null);
  const animationFrameRef = useRef<number | null>(null);

  // Web Audio refs for spectral visualizer
  const audioContextRef = useRef<AudioContext | null>(null);
  const analyserRef = useRef<AnalyserNode | null>(null);
  const sourceRef = useRef<MediaElementAudioSourceNode | null>(null);
  const handleTrackEndedRef = useRef<() => void>(() => {});

  // Initialize analyser node
  const initAnalyser = () => {
    if (!audioRef.current) return;
    if (analyserRef.current) return;

    try {
      const AudioContextClass = window.AudioContext || (window as any).webkitAudioContext;
      if (!AudioContextClass) return;

      const ctx = new AudioContextClass();
      const analyser = ctx.createAnalyser();
      analyser.fftSize = 128; // gives 64 frequency bins, perfect for our spectrum
      analyser.smoothingTimeConstant = 0.75;

      const source = ctx.createMediaElementSource(audioRef.current);
      source.connect(analyser);
      analyser.connect(ctx.destination);

      audioContextRef.current = ctx;
      analyserRef.current = analyser;
      sourceRef.current = source;
    } catch (err) {
      console.warn('Web Audio API is not supported or failed to init:', err);
    }
  };

  // Initialize Audio
  useEffect(() => {
    const audio = new Audio();
    audio.preload = 'auto';
    audio.crossOrigin = 'anonymous'; // Important for CORS security with AudioContext
    audioRef.current = audio;

    const handleTimeUpdate = () => {
      setState((prev) => ({ ...prev, currentTime: audio.currentTime }));
    };

    const handleDurationChange = () => {
      if (audio.duration && !isNaN(audio.duration)) {
        setState((prev) => ({ ...prev, duration: audio.duration }));
      }
    };

    const handleLoadedMetadata = () => {
      if (audio.duration && !isNaN(audio.duration)) {
        setState((prev) => ({ ...prev, duration: audio.duration }));
      }
    };

    const handleEnded = () => {
      handleTrackEndedRef.current();
    };

    const handleError = () => {
      console.warn('Audio loading failed, switching to simulated playback');
      // If a track fails, we don't crash, we let it simulate so the visualizer and timers still animate
    };

    audio.addEventListener('timeupdate', handleTimeUpdate);
    audio.addEventListener('durationchange', handleDurationChange);
    audio.addEventListener('loadedmetadata', handleLoadedMetadata);
    audio.addEventListener('ended', handleEnded);
    audio.addEventListener('error', handleError);

    return () => {
      audio.removeEventListener('timeupdate', handleTimeUpdate);
      audio.removeEventListener('durationchange', handleDurationChange);
      audio.removeEventListener('loadedmetadata', handleLoadedMetadata);
      audio.removeEventListener('ended', handleEnded);
      audio.removeEventListener('error', handleError);
      audio.pause();
      if (animationFrameRef.current) {
        cancelAnimationFrame(animationFrameRef.current);
      }
    };
  }, []);

  const currentTrack = tracks[state.currentTrackIndex] || tracks[0];

  // Fetch local music folder tracks on mount
  useEffect(() => {
    const loadLocalTracks = async () => {
      const fallbackTracks: Track[] = [
        {
          id: "github-0-chahun-main-ya-naa",
          title: "Chahun Main Ya Naa",
          artist: "Bhoomika",
          album: "Aura Sessions",
          duration: "4:00",
          durationSec: 240,
          coverUrl: "https://images.unsplash.com/photo-1514525253161-7a46d19cd819?q=80&w=500&auto=format&fit=crop",
          audioUrl: "https://raw.githubusercontent.com/Ronit-CodeWizard/wizard-resources/main/Musics/BHO/Chahun%20Main%20Ya%20Naa%20-%20BHO.mp3",
        },
        {
          id: "github-1-ishq-mubarak",
          title: "Ishq Mubarak",
          artist: "Bhoomika",
          album: "Aura Sessions",
          duration: "4:00",
          durationSec: 240,
          coverUrl: "https://images.unsplash.com/photo-1498038432885-c6f3f1b912ee?q=80&w=500&auto=format&fit=crop",
          audioUrl: "https://raw.githubusercontent.com/Ronit-CodeWizard/wizard-resources/main/Musics/BHO/Ishq%20Mubarak%20-%20BHO.mp3",
        },
        {
          id: "github-2-jeena-jeena",
          title: "Jeena Jeena",
          artist: "Bhoomika",
          album: "Aura Sessions",
          duration: "4:00",
          durationSec: 240,
          coverUrl: "https://images.unsplash.com/photo-1470225620780-dba8ba36b745?q=80&w=500&auto=format&fit=crop",
          audioUrl: "https://raw.githubusercontent.com/Ronit-CodeWizard/wizard-resources/main/Musics/BHO/Jeena%20Jeena%20-%20BHO.mp3",
        },
        {
          id: "github-3-nazar-na-lag-jaaye",
          title: "Nazar Na Lag Jaaye",
          artist: "Bhoomika",
          album: "Aura Sessions",
          duration: "4:00",
          durationSec: 240,
          coverUrl: "https://images.unsplash.com/photo-1511671782779-c97d3d27a1d4?q=80&w=500&auto=format&fit=crop",
          audioUrl: "https://raw.githubusercontent.com/Ronit-CodeWizard/wizard-resources/main/Musics/BHO/Nazar%20Na%20Lag%20Jaaye%20-%20BHO.mp3",
        },
        {
          id: "github-4-saiyaara",
          title: "Saiyaara",
          artist: "Bhoomika",
          album: "Aura Sessions",
          duration: "4:00",
          durationSec: 240,
          coverUrl: "https://images.unsplash.com/photo-1511379938547-c1f69419868d?q=80&w=500&auto=format&fit=crop",
          audioUrl: "https://raw.githubusercontent.com/Ronit-CodeWizard/wizard-resources/main/Musics/BHO/Saiyaara%20-%20BHO.mp3",
        },
        {
          id: "github-5-tum-ho-toh",
          title: "Tum Ho Toh",
          artist: "Bhoomika",
          album: "Aura Sessions",
          duration: "4:00",
          durationSec: 240,
          coverUrl: "https://images.unsplash.com/photo-1501386761578-eac5c94b800a?q=80&w=500&auto=format&fit=crop",
          audioUrl: "https://raw.githubusercontent.com/Ronit-CodeWizard/wizard-resources/main/Musics/BHO/Tum%20Ho%20Toh%20-%20BHO.mp3",
        },
      ];

      let fetchedTracks: Track[] = [];

      try {
        // Step 1: Attempt to fetch from local API endpoint
        const response = await fetch('/api/tracks');
        if (response.ok) {
          const contentType = response.headers.get('content-type');
          if (contentType && contentType.includes('application/json')) {
            const localTracks = await response.json() as Track[];
            if (localTracks && localTracks.length > 0) {
              fetchedTracks = localTracks;
            }
          }
        }
      } catch (err) {
        console.warn('Local tracks endpoint not reachable, fetching directly from GitHub...', err);
      }

      // Step 2: If local endpoint didn't return tracks, fetch directly from GitHub
      if (fetchedTracks.length === 0) {
        try {
          const ghResponse = await fetch('https://api.github.com/repos/Ronit-CodeWizard/wizard-resources/contents/Musics/BHO');
          if (ghResponse.ok) {
            const files = await ghResponse.json() as any[];
            if (Array.isArray(files) && files.length > 0) {
              const audioFiles = files.filter(
                (f) => f.type === 'file' && f.name.toLowerCase().endsWith('.mp3')
              );

              const coverImages = [
                "https://images.unsplash.com/photo-1514525253161-7a46d19cd819?q=80&w=500&auto=format&fit=crop",
                "https://images.unsplash.com/photo-1498038432885-c6f3f1b912ee?q=80&w=500&auto=format&fit=crop",
                "https://images.unsplash.com/photo-1470225620780-dba8ba36b745?q=80&w=500&auto=format&fit=crop",
                "https://images.unsplash.com/photo-1511671782779-c97d3d27a1d4?q=80&w=500&auto=format&fit=crop",
                "https://images.unsplash.com/photo-1511379938547-c1f69419868d?q=80&w=500&auto=format&fit=crop",
                "https://images.unsplash.com/photo-1501386761578-eac5c94b800a?q=80&w=500&auto=format&fit=crop"
              ];

              fetchedTracks = audioFiles.map((file, index) => {
                let title = file.name.replace(/\.[^/.]+$/, ""); // Strip extension
                if (title.includes(" - ")) {
                  title = title.split(" - ")[0];
                } else if (title.includes("-")) {
                  title = title.split("-")[0];
                }
                title = title.trim();

                return {
                  id: `github-${index}-${encodeURIComponent(file.name)}`,
                  title: title || "Untitled Track",
                  artist: "Bhoomika",
                  album: "Aura Sessions",
                  duration: "4:00",
                  durationSec: 240,
                  coverUrl: coverImages[index % coverImages.length],
                  audioUrl: file.download_url || `https://raw.githubusercontent.com/Ronit-CodeWizard/wizard-resources/main/Musics/BHO/${encodeURIComponent(file.name)}`,
                };
              });
            }
          }
        } catch (ghErr) {
          console.warn('Direct GitHub fetch failed, loading static premium fallbacks', ghErr);
        }
      }

      // Step 3: Use static tracks if both failed
      if (fetchedTracks.length === 0) {
        fetchedTracks = fallbackTracks;
      }

      // Restore custom duration / info from localStorage
      const savedStr = localStorage.getItem('aura_tracks');
      let savedTracks: Track[] = [];
      if (savedStr) {
        try {
          savedTracks = JSON.parse(savedStr);
        } catch (e) {}
      }

      const mergedTracks = fetchedTracks.map((track) => {
        const saved = savedTracks.find((t) => t.id === track.id);
        if (saved) {
          return {
            ...track,
            duration: saved.duration || track.duration,
            durationSec: saved.durationSec || track.durationSec,
          };
        }
        return track;
      });

      setTracks(mergedTracks);

      // Preload metadata to resolve correct audio duration in the background
      mergedTracks.forEach((track) => {
        if (track.audioUrl && (!track.durationSec || track.durationSec === 240)) {
          const tempAudio = new Audio();
          tempAudio.src = track.audioUrl;
          tempAudio.preload = 'metadata';
          tempAudio.addEventListener('loadedmetadata', () => {
            if (tempAudio.duration && !isNaN(tempAudio.duration)) {
              const durationSec = tempAudio.duration;
              const mins = Math.floor(durationSec / 60);
              const secs = Math.floor(durationSec % 60);
              const durationStr = `${mins}:${secs < 10 ? '0' : ''}${secs}`;

              setTracks((prevTracks) => {
                const updated = [...prevTracks];
                const trackIdx = updated.findIndex((t) => t.id === track.id);
                if (trackIdx !== -1) {
                  updated[trackIdx] = {
                    ...updated[trackIdx],
                    duration: durationStr,
                    durationSec: Math.floor(durationSec),
                  };
                }
                return updated;
              });
            }
          });
        }
      });
    };

    loadLocalTracks();
  }, []);

  // Update audio source when track index changes
  useEffect(() => {
    if (!audioRef.current || !currentTrack) return;

    const audio = audioRef.current;
    const wasPlaying = state.isPlaying;

    audio.src = currentTrack.audioUrl;
    audio.load();
    
    // Reset timer
    setState((prev) => ({ 
      ...prev, 
      currentTime: 0, 
      duration: currentTrack.durationSec || 240 
    }));

    if (wasPlaying) {
      setTimeout(() => {
        playAudio(true); // Pass true to register play counter for new song
      }, 50);
    }
  }, [state.currentTrackIndex]);

  // Update volume
  useEffect(() => {
    if (!audioRef.current) return;
    audioRef.current.volume = state.isMuted ? 0 : state.volume;
    localStorage.setItem('aura_volume', state.volume.toString());
  }, [state.volume, state.isMuted]);

  // Save liked tracks
  useEffect(() => {
    localStorage.setItem('aura_liked', JSON.stringify(state.likedTrackIds));
  }, [state.likedTrackIds]);

  // Save track ordering
  useEffect(() => {
    localStorage.setItem('aura_tracks', JSON.stringify(tracks));
  }, [tracks]);

  // Track listening time in real-time
  useEffect(() => {
    if (!state.isPlaying) return;
    const interval = setInterval(() => {
      setTimeStreamedSec((prev) => prev + 1);
    }, 1000);
    return () => clearInterval(interval);
  }, [state.isPlaying]);

  // Simulation fallback for browser-blocked autoplay or dead links
  const simulationTimerRef = useRef<NodeJS.Timeout | null>(null);

  const startSimulation = () => {
    if (simulationTimerRef.current) clearInterval(simulationTimerRef.current);
    simulationTimerRef.current = setInterval(() => {
      setState((prev) => {
        const nextTime = prev.currentTime + 1;
        if (nextTime >= prev.duration) {
          clearInterval(simulationTimerRef.current!);
          setTimeout(() => handleTrackEndedRef.current(), 100);
          return { ...prev, currentTime: 0 };
        }
        return { ...prev, currentTime: nextTime };
      });
    }, 1000);
  };

  const stopSimulation = () => {
    if (simulationTimerRef.current) {
      clearInterval(simulationTimerRef.current);
      simulationTimerRef.current = null;
    }
  };

  const playAudio = async (isNewTrack = false) => {
    if (!audioRef.current) return;
    setPlayError(null);
    try {
      // Initialize analyser on user gesture
      initAnalyser();

      if (audioContextRef.current && audioContextRef.current.state === 'suspended') {
        await audioContextRef.current.resume();
      }

      stopSimulation();
      if (isNewTrack || !state.isPlaying) {
        setTotalPlays((p) => p + 1);
      }
      setState((prev) => ({ ...prev, isPlaying: true }));
      await audioRef.current.play();
    } catch (err: any) {
      console.warn('Playback block or issue:', err);
      if (err.name === 'NotAllowedError') {
        setPlayError('Interaction required to play audio');
      }
      // Start simulated playback so the circular vinyl and seekbar still animate!
      if (isNewTrack || !state.isPlaying) {
        setTotalPlays((p) => p + 1);
      }
      setState((prev) => ({ ...prev, isPlaying: true }));
      startSimulation();
    }
  };

  const pauseAudio = () => {
    if (!audioRef.current) return;
    audioRef.current.pause();
    stopSimulation();
    setState((prev) => ({ ...prev, isPlaying: false }));
  };

  const togglePlay = () => {
    if (state.isPlaying) {
      pauseAudio();
    } else {
      playAudio();
    }
  };

  const playTrack = (index: number, forcePlay = false) => {
    if (index < 0 || index >= tracks.length) return;
    if (index === state.currentTrackIndex && !forcePlay) {
      if (!state.isPlaying) {
        playAudio();
      } else {
        pauseAudio();
      }
    } else {
      setState((prev) => ({ ...prev, currentTrackIndex: index, isPlaying: true }));
      if (index === state.currentTrackIndex && forcePlay) {
        if (audioRef.current) {
          audioRef.current.src = tracks[index].audioUrl;
          audioRef.current.load();
        }
        setState((prev) => ({ 
          ...prev, 
          currentTime: 0, 
          duration: tracks[index].durationSec || 240,
          isPlaying: true
        }));
        setTimeout(() => {
          playAudio(true);
        }, 50);
      }
    }
  };

  const next = () => {
    if (tracks.length === 0) return;
    if (state.queue.length > 0) {
      const nextTrackId = state.queue[0];
      const nextIndex = tracks.findIndex((t) => t.id === nextTrackId);
      setState((prev) => ({
        ...prev,
        queue: prev.queue.slice(1),
        currentTrackIndex: nextIndex !== -1 ? nextIndex : (prev.currentTrackIndex + 1) % tracks.length,
        isPlaying: true,
      }));
      return;
    }

    if (state.isShuffle) {
      if (tracks.length <= 1) {
        playTrack(0, true);
      } else {
        let randIndex = state.currentTrackIndex;
        while (randIndex === state.currentTrackIndex) {
          randIndex = Math.floor(Math.random() * tracks.length);
        }
        playTrack(randIndex, true);
      }
    } else {
      const nextIndex = (state.currentTrackIndex + 1) % tracks.length;
      playTrack(nextIndex, true);
    }
  };

  const prev = () => {
    if (tracks.length === 0) return;
    if (state.isShuffle) {
      if (tracks.length <= 1) {
        playTrack(0, true);
      } else {
        let randIndex = state.currentTrackIndex;
        while (randIndex === state.currentTrackIndex) {
          randIndex = Math.floor(Math.random() * tracks.length);
        }
        playTrack(randIndex, true);
      }
    } else {
      const prevIndex = (state.currentTrackIndex - 1 + tracks.length) % tracks.length;
      playTrack(prevIndex, true);
    }
  };

  const handleTrackEnded = () => {
    if (state.isRepeat === 'one') {
      if (audioRef.current) {
        audioRef.current.currentTime = 0;
        playAudio();
      }
    } else if (state.isRepeat === 'all') {
      next();
    } else {
      // none
      if (state.currentTrackIndex < tracks.length - 1) {
        next();
      } else {
        pauseAudio();
      }
    }
  };
  handleTrackEndedRef.current = handleTrackEnded;

  const seek = (time: number) => {
    if (!audioRef.current) return;
    const clampedTime = Math.max(0, Math.min(time, state.duration));
    audioRef.current.currentTime = clampedTime;
    setState((prev) => ({ ...prev, currentTime: clampedTime }));
  };

  const setVolume = (vol: number) => {
    const clampedVol = Math.max(0, Math.min(vol, 1));
    setState((prev) => ({ ...prev, volume: clampedVol, isMuted: clampedVol === 0 }));
  };

  const toggleMute = () => {
    setState((prev) => ({ ...prev, isMuted: !prev.isMuted }));
  };

  const toggleShuffle = () => {
    setState((prev) => ({ ...prev, isShuffle: !prev.isShuffle }));
  };

  const toggleRepeat = () => {
    setState((prev) => {
      let nextRepeat: 'none' | 'all' | 'one' = 'none';
      if (prev.isRepeat === 'none') nextRepeat = 'all';
      else if (prev.isRepeat === 'all') nextRepeat = 'one';
      return { ...prev, isRepeat: nextRepeat };
    });
  };

  const toggleLike = (trackId: string) => {
    setState((prev) => {
      const isLiked = prev.likedTrackIds.includes(trackId);
      const likedTrackIds = isLiked
        ? prev.likedTrackIds.filter((id) => id !== trackId)
        : [...prev.likedTrackIds, trackId];
      return { ...prev, likedTrackIds };
    });
  };

  const reorderPlaylist = (startIndex: number, endIndex: number) => {
    if (startIndex < 0 || startIndex >= tracks.length || endIndex < 0 || endIndex >= tracks.length) return;
    const result = [...tracks];
    const [removed] = result.splice(startIndex, 1);
    result.splice(endIndex, 0, removed);
    
    // Adjust current playing index to maintain playing song
    const currentPlayingTrack = tracks[state.currentTrackIndex];
    const newPlayingIndex = result.findIndex((t) => t.id === currentPlayingTrack.id);

    setTracks(result);
    setState((prev) => ({
      ...prev,
      currentTrackIndex: newPlayingIndex !== -1 ? newPlayingIndex : 0,
    }));
  };

  const addToQueue = (trackId: string) => {
    setState((prev) => {
      if (prev.queue.includes(trackId)) return prev;
      return { ...prev, queue: [...prev.queue, trackId] };
    });
  };

  const removeFromQueue = (trackId: string) => {
    setState((prev) => ({
      ...prev,
      queue: prev.queue.filter((id) => id !== trackId),
    }));
  };

  return {
    tracks,
    currentTrack,
    state,
    playError,
    audioElement: audioRef.current,
    analyserNode: analyserRef.current,
    totalPlays,
    timeStreamedSec,
    togglePlay,
    playTrack,
    next,
    prev,
    seek,
    setVolume,
    toggleMute,
    toggleShuffle,
    toggleRepeat,
    toggleLike,
    reorderPlaylist,
    addToQueue,
    removeFromQueue,
  };
}
export type UseAudioPlayerType = ReturnType<typeof useAudioPlayer>;
